# 3. Nodos: nombre:RAM_MB:vCPU:último octeto de IP
NODOS="nodo-nginx:2048:1:11 nodo-so:2048:1:12 nodo-postgres:2048:1:13 nodo-tomcat:2048:1:14 nodo-jboss:4096:2:15"
for n in $NODOS; do
  IFS=: read -r NAME RAM CPU OCT <<<"$n"
  MAC="52:54:00:de:00:$OCT"
  IP="192.168.122.$OCT"
  if sudo virsh dominfo "$NAME" >/dev/null 2>&1; then log "   $NAME ya existe"; continue; fi
  sudo virsh net-update default add ip-dhcp-host "<host mac='$MAC' name='$NAME' ip='$IP'/>" --live --config >/dev/null 2>&1 || true
  sudo qemu-img create -q -f qcow2 -F qcow2 -b "$IMG_DIR/rhel9-base.qcow2" "$IMG_DIR/$NAME.qcow2" 20G
  UD="$(mktemp)"
  cat > "$UD" <<EOF
#cloud-config
hostname: $NAME
fqdn: $NAME.demo.local
users:
  - name: ansible
    sudo: ALL=(ALL) NOPASSWD:ALL
    shell: /bin/bash
    ssh_authorized_keys:
      - $PUB
ssh_pwauth: false
EOF
  sudo virt-install --name "$NAME" --memory "$RAM" --vcpus "$CPU" --import \
    --disk "$IMG_DIR/$NAME.qcow2,bus=virtio" --osinfo linux2022 \
    --network "network=default,mac=$MAC,model=virtio" --graphics none \
    --cloud-init "user-data=$UD" --noautoconsole >>"$LOG" 2>&1 || fail "No se pudo crear $NAME"
  sudo virsh autostart "$NAME" >/dev/null
  rm -f "$UD"
  grep -q " $NAME$" /etc/hosts || echo "$IP $NAME.demo.local $NAME" | sudo tee -a /etc/hosts >/dev/null
  log "   $NAME creado ($IP, ${RAM} MB)"
done
cat > "$HOME/.ssh/config.demo" <<'EOF'
Host nodo-*
  User ansible
  IdentityFile ~/.ssh/id_demo
  StrictHostKeyChecking accept-new
EOF
grep -q config.demo "$HOME/.ssh/config" 2>/dev/null || { touch "$HOME/.ssh/config"; sed -i '1i Include ~/.ssh/config.demo' "$HOME/.ssh/config"; chmod 600 "$HOME/.ssh/config"; }
for i in $(seq 1 60); do
  OK=0
  for n in $NODOS; do NAME="${n%%:*}"; ssh -o ConnectTimeout=3 -o BatchMode=yes "$NAME" true 2>/dev/null && OK=$((OK+1)); done
  [ "$OK" -eq 5 ] && break
  sleep 10
done
log "3/4 Nodos accesibles por SSH: $OK de 5"
