# Registra los nodos de la demo en Red Hat para habilitar repositorios.
- name: Registrar nodos en Red Hat
  hosts: nodos
  become: true
  gather_facts: false
  vars:
    org_id: "12665691"
  vars_prompt:
    - name: clave_activacion
      prompt: "Activation key"
      private: true
  tasks:
    # subscription-manager no tiene módulo en ansible-core
    - name: Estado del registro
      ansible.builtin.command: subscription-manager identity
      register: r_id
      changed_when: false
      failed_when: false

    - name: Registrar con activation key
      ansible.builtin.command: >-
        subscription-manager register --org={{ org_id }} --activationkey={{ clave_activacion }}
      when: r_id.rc != 0
      register: r_reg
      failed_when: false
      no_log: true

    - name: Resultado del registro
      ansible.builtin.debug:
        msg: "{{ (r_reg.stderr | default('')) ~ ' ' ~ (r_reg.stdout | default('')) }}"
